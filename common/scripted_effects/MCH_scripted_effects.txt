	if = {
		limit = {
			has_idea = MCH_cristeros_backed_insurgents_4
		}
		remove_ideas = MCH_cristeros_backed_insurgents_4
		set_country_flag = MCH_dealt_with_insurgency_flag
	}
	if = {
		limit = {
			has_idea = MCH_cristeros_backed_insurgents_3
		}
		swap_ideas = {
			remove_idea = MCH_cristeros_backed_insurgents_3
			add_idea = MCH_cristeros_backed_insurgents_4
		}
	}
	if = {
		limit = {
			has_idea = MCH_cristeros_backed_insurgents_2
		}
		swap_ideas = {
			remove_idea = MCH_cristeros_backed_insurgents_2
			add_idea = MCH_cristeros_backed_insurgents_3
		}
	}
	if = {
		limit = {
			has_idea = MCH_cristeros_backed_insurgents
		}
		swap_ideas = {
			remove_idea = MCH_cristeros_backed_insurgents
			add_idea = MCH_cristeros_backed_insurgents_2
		}
	}
}
