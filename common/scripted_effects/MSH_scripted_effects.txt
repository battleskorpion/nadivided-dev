church_power_lvl_up = {
	if = {
		limit = {
			has_idea = MSH_church_power_4
		}
		swap_ideas = {
			remove_idea = MSH_church_power_4
			add_idea = MSH_church_power_5
		}
	}
	if = {
		limit = {
			has_idea = MSH_church_power_3
		}
		swap_ideas = {
			remove_idea = MSH_church_power_3
			add_idea = MSH_church_power_4
		}
	}
	if = {
		limit = {
			has_idea = MSH_church_power_2
		}
		swap_ideas = {
			remove_idea = MSH_church_power_2
			add_idea = MSH_church_power_3
		}
	}
	if = {
		limit = {
			has_idea = MSH_church_power_1
		}
		swap_ideas = {
			remove_idea = MSH_church_power_1
			add_idea = MSH_church_power_2
		}
	}
}

church_power_lvl_down = {
	if = {
		limit = {
			has_idea = MSH_church_power_2
		}
		swap_ideas = {
			remove_idea = MSH_church_power_2
			add_idea = MSH_church_power_1
		}
	}
	if = {
		limit = {
			has_idea = MSH_church_power_3
		}
		swap_ideas = {
			remove_idea = MSH_church_power_3
			add_idea = MSH_church_power_2
		}
	}
	if = {
		limit = {
			has_idea = MSH_church_power_4
		}
		swap_ideas = {
			remove_idea = MSH_church_power_4
			add_idea = MSH_church_power_3
		}
	}
	if = {
		limit = {
			has_idea = MSH_church_power_5
		}
		swap_ideas = {
			remove_idea = MSH_church_power_5
			add_idea = MSH_church_power_4
		}
	}
}
