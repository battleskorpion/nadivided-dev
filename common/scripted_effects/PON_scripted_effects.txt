PON_upgrade_canada_idea = {
	if = {
		limit = {
			has_idea = PON_ontario_citizen_act_canada_4
		}
		swap_ideas = {
			remove_idea = PON_ontario_citizen_act_canada_4
			add_idea = PON_ontario_citizen_act_canada_5
		}
	}
	if = {
		limit = {
			has_idea = PON_ontario_citizen_act_canada_3
		}
		swap_ideas = {
			remove_idea = PON_ontario_citizen_act_canada_3
			add_idea = PON_ontario_citizen_act_canada_4
		}
	}
	if = {
		limit = {
			has_idea = PON_ontario_citizen_act_canada_2
		}
		swap_ideas = {
			remove_idea = PON_ontario_citizen_act_canada_2
			add_idea = PON_ontario_citizen_act_canada_3
		}
	}
	if = {
		limit = {
			has_idea = PON_ontario_citizen_act_canada
		}
		swap_ideas = {
			remove_idea = PON_ontario_citizen_act_canada
			add_idea = PON_ontario_citizen_act_canada_2
		}
	}
}
